module ultrasonic(
    input  wire clk,        // FPGA clock
    input  wire echo,
    output reg  trig,
    output reg [31:0] distance
);

reg [31:0] counter = 0;
reg [31:0] echo_count = 0;
reg measuring = 0;

always @(posedge clk) begin

    // Generate Trigger Pulse
    if(counter < 500)          // Adjust based on clock frequency
        trig <= 1;
    else
        trig <= 0;

    if(counter == 5000000)
        counter <= 0;
    else
        counter <= counter + 1;

    // Measure Echo Pulse Width
    if(echo && !measuring) begin
        measuring <= 1;
        echo_count <= 0;
    end

    if(measuring && echo)
        echo_count <= echo_count + 1;

    if(measuring && !echo) begin
        measuring <= 0;
        distance <= echo_count;
    end

end

endmodule
